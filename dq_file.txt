import math


def _is_valid_number(value):
    return (
        value is not None
        and not (
            isinstance(value, float)
            and math.isnan(value)
        )
    )


def explain_anomaly(
    row,
    z_score_threshold=3.0,
    null_rate_threshold=0.0,
    duplicate_rate_threshold=0.0,
):
    reasons = []

    # --------------------------------------------------
    # NULL RATE
    # --------------------------------------------------

    if "null_rate" in row.index:

        null_rate = row["null_rate"]

        if _is_valid_number(null_rate):

            if null_rate > null_rate_threshold:

                reasons.append(
                    f"High NULL rate: "
                    f"{null_rate:.2%} "
                    f"(threshold: "
                    f"{null_rate_threshold:.2%})"
                )

    # --------------------------------------------------
    # DUPLICATE RATE
    # --------------------------------------------------

    if "duplicate_rate" in row.index:

        duplicate_rate = row["duplicate_rate"]

        if _is_valid_number(duplicate_rate):

            if duplicate_rate > duplicate_rate_threshold:

                reasons.append(
                    f"High duplicate rate: "
                    f"{duplicate_rate:.2%} "
                    f"(threshold: "
                    f"{duplicate_rate_threshold:.2%})"
                )

    # --------------------------------------------------
    # RECORD COUNT / VOLUME
    # --------------------------------------------------

    if "record_count" in row.index:

        record_count = row["record_count"]

        if _is_valid_number(record_count):

            # If a historical/expected count exists,
            # compare against it.
            if (
                "expected_record_count" in row.index
                and _is_valid_number(
                    row["expected_record_count"]
                )
            ):

                expected = row[
                    "expected_record_count"
                ]

                if record_count < expected:

                    percentage = (
                        (expected - record_count)
                        / expected
                        * 100
                        if expected > 0
                        else 0
                    )

                    reasons.append(
                        f"Low record volume: "
                        f"{record_count:.0f} records "
                        f"vs expected "
                        f"{expected:.0f} "
                        f"({percentage:.1f}% below expected)"
                    )

    # --------------------------------------------------
    # Z-SCORE
    # --------------------------------------------------

    if "z_score" in row.index:

        z_score = row["z_score"]

        if _is_valid_number(z_score):

            if abs(z_score) >= z_score_threshold:

                direction = (
                    "above"
                    if z_score > 0
                    else "below"
                )

                reasons.append(
                    f"Statistical outlier: "
                    f"z-score {z_score:.2f} "
                    f"({direction} normal range; "
                    f"threshold ±{z_score_threshold:.2f})"
                )

    # --------------------------------------------------
    # MISSING HOUR
    # --------------------------------------------------

    if "record_count" in row.index:

        record_count = row["record_count"]

        if (
            _is_valid_number(record_count)
            and record_count == 0
        ):

            reasons.append(
                "Missing data: "
                "no records received for this time window"
            )

    # --------------------------------------------------
    # DATA COMPLETENESS
    # --------------------------------------------------

    if "completeness_rate" in row.index:

        completeness_rate = row[
            "completeness_rate"
        ]

        if _is_valid_number(completeness_rate):

            if completeness_rate < 1.0:

                reasons.append(
                    f"Low data completeness: "
                    f"{completeness_rate:.2%}"
                )

    # --------------------------------------------------
    # FALLBACK
    # --------------------------------------------------

    if not reasons:

        reasons.append(
            "ML model detected an unusual "
            "combination of data-quality metrics"
        )

    return reasons
