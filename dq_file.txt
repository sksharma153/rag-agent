import math


def is_valid_number(value):
    if value is None:
        return False

    try:
        return not math.isnan(float(value))
    except (TypeError, ValueError):
        return False


def format_percentage(value):
    return f"{float(value) * 100:.2f}%"


def explain_anomaly(
    row,
    z_score_threshold=3.0,
    null_rate_threshold=0.0,
    duplicate_rate_threshold=0.0,
):
    reasons = []

    # ============================================================
    # 1. MISSING HOUR
    # ============================================================

    if "is_missing_hour" in row.index:

        value = row["is_missing_hour"]

        if value in [True, 1, "true", "True"]:

            reasons.append(
                "Missing hourly data: no data was received "
                "for the expected hour"
            )

    # ============================================================
    # 2. Z-SCORE BASED ANOMALIES
    # ============================================================

    zscore_columns = [
        column
        for column in row.index
        if column.endswith("_zscore")
    ]

    zscore_anomalies = []

    for column in zscore_columns:

        value = row[column]

        if not is_valid_number(value):
            continue

        value = float(value)

        if abs(value) >= z_score_threshold:

            zscore_anomalies.append(
                (column, value)
            )

    # Sort strongest statistical deviations first

    zscore_anomalies.sort(
        key=lambda x: abs(x[1]),
        reverse=True,
    )

    for column, value in zscore_anomalies[:5]:

        metric_name = (
            column
            .replace("_zscore", "")
            .replace("_", " ")
            .title()
        )

        direction = (
            "higher"
            if value > 0
            else "lower"
        )

        reasons.append(
            f"Statistical outlier: {metric_name} "
            f"is {direction} than normal "
            f"(z-score: {value:.2f}, "
            f"threshold: ±{z_score_threshold:.2f})"
        )

    # ============================================================
    # 3. NULL RATIOS
    # ============================================================

    null_ratio_columns = [
        column
        for column in row.index
        if column.endswith("_null_ratio")
    ]

    for column in null_ratio_columns:

        value = row[column]

        if not is_valid_number(value):
            continue

        value = float(value)

        if value > null_rate_threshold:

            field_name = (
                column
                .replace("_null_ratio", "")
                .replace("_", " ")
                .title()
            )

            reasons.append(
                f"NULL values detected in "
                f"{field_name} fields: "
                f"{format_percentage(value)}"
            )

    # ============================================================
    # 4. CONDITIONAL VIOLATION
    # ============================================================

    if "conditional_violation_ratio" in row.index:

        value = row[
            "conditional_violation_ratio"
        ]

        if is_valid_number(value):

            value = float(value)

            if value > 0:

                reasons.append(
                    f"Conditional data-quality violations: "
                    f"{format_percentage(value)} of records"
                )

    # ============================================================
    # 5. BUSINESS / DATA QUALITY FLAGS
    # ============================================================

    flag_messages = {

        "flag_missing_all_items":
            "Missing all order items",

        "flag_missing_all_passengers":
            "Missing all passenger information",

        "flag_missing_all_services":
            "Missing all service information",

        "flag_cancelled_missing_reason":
            "Cancelled order is missing cancellation reason",

        "flag_item_amt_without_ccy":
            "Item amount exists without currency",

        "flag_item_base_amt_without_ccy":
            "Item base amount exists without currency",

        "flag_active_svc_missing_ticket":
            "Active service is missing ticket information",

        "flag_flight_missing_times":
            "Flight is missing required timing information",

        "flag_flight_missing_legs":
            "Flight is missing required leg information",

        "flag_pay_amt_without_ccy":
            "Payment amount exists without currency",
    }

    for column, message in flag_messages.items():

        if column not in row.index:
            continue

        value = row[column]

        if value in [True, 1, "true", "True"]:

            reasons.append(
                f"Data-quality rule violation: "
                f"{message}"
            )

    # ============================================================
    # 6. VOLUME RELATED ANOMALIES
    # ============================================================

    volume_columns = [
        "row_count_score",
        "event_count_zscore",
        "active_pax_zscore",
        "order_total_zscore",
    ]

    for column in volume_columns:

        if column not in row.index:
            continue

        value = row[column]

        if not is_valid_number(value):
            continue

        value = float(value)

        if abs(value) >= z_score_threshold:

            metric_name = (
                column
                .replace("_zscore", "")
                .replace("_score", "")
                .replace("_", " ")
                .title()
            )

            direction = (
                "higher"
                if value > 0
                else "lower"
            )

            reasons.append(
                f"Volume anomaly: {metric_name} "
                f"is {direction} than expected "
                f"(score: {value:.2f})"
            )

    # ============================================================
    # 7. REMOVE DUPLICATE EXPLANATIONS
    # ============================================================

    unique_reasons = []

    for reason in reasons:

        if reason not in unique_reasons:
            unique_reasons.append(reason)

    reasons = unique_reasons

    # ============================================================
    # 8. FALLBACK
    # ============================================================

    if not reasons:

        anomaly_score = row.get(
            "anomaly_score"
        )

        if is_valid_number(anomaly_score):

            reasons.append(
                "Isolation Forest detected an unusual "
                "combination of data-quality features "
                f"(anomaly score: {float(anomaly_score):.4f})"
            )

        else:

            reasons.append(
                "Isolation Forest detected an unusual "
                "combination of data-quality features"
            )

    return reasons
